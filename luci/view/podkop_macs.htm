<%:Sync status%>

<%
local fs = require("nixio.fs")
local sys = require("luci.sys")

local function status_badge(state)
    if state == "OK" then
        return 'OK'
    end
    if state == "WARN" then
        return 'WARN'
    end
    return 'FAIL'
end

local dnsmasq_conf = fs.readfile("/etc/dnsmasq.conf") or ""
local has_dhcp_hook = dnsmasq_conf:find("dhcp%-script=/usr/bin/podkop%-sync%-excluded", 1, false) ~= nil
local has_script = fs.access("/usr/bin/podkop-sync-excluded")
local script_exec = fs.access("/usr/bin/podkop-sync-excluded", "x")

local cron_tab = fs.readfile("/etc/crontabs/root") or ""
local has_cron = cron_tab:find("podkop%-sync%-excluded", 1, false) ~= nil

local dhcp_hook_state = "FAIL"
if has_dhcp_hook then
    dhcp_hook_state = "OK"
elseif has_cron then
    dhcp_hook_state = "WARN"
end

local cron_state = "FAIL"
if has_cron then
    cron_state = "OK"
elseif has_dhcp_hook then
    cron_state = "WARN"
end

local last_log = sys.exec("logread -e 'podkop-sync' | tail -n 1 2>/dev/null")
last_log = (last_log or ""):gsub("%s+$", "")
if last_log == "" then
    last_log = "[no podkop-sync entries yet]"
end

luci.http.write(string.format("%-32s %s\n", translate("dhcp-script hook in dnsmasq"), status_badge(dhcp_hook_state)))
luci.http.write(string.format("%-32s %s\n", translate("sync script file exists"), status_badge(has_script and "OK" or "FAIL")))
luci.http.write(string.format("%-32s %s\n", translate("sync script is executable"), status_badge(script_exec and "OK" or "FAIL")))
luci.http.write(string.format("%-32s %s\n", translate("cron fallback enabled"), status_badge(cron_state)))
luci.http.write("\n" .. translate("Last podkop-sync log entry") .. ":\n" .. last_log .. "\n")
%>

            hostname = hostname or ""
            luci.http.write(string.format("%-18s  %-15s  %s\n", mac, ip, hostname))
        end
    end
    leases:close()
